01
مسألة لوحتي الإعلانات
🔵 هندسة المستطيلات · تقاطع المساحات
قامت إحدى الشركات بوضع لوحتي إعلانات في ساحة المدينة (على شكل مستطيلات). اللوحة الأولى إحداثياتها (x1, y1) للزاوية السفلى اليسرى و (x2, y2) للزاوية العليا اليمنى. واللوحة الثانية إحداثياتها (x3, y3) و (x4, y4). أوجد المساحة الإجمالية التي تغطيها اللوحتان معاً لكي يتم حساب تكلفة الإيجار.

⚠️ تلميح: احذر من حساب المنطقة المتقاطعة بينهما مرتين!
💡 الشرح المنطقي
لحساب المساحة الإجمالية، نحسب مساحة اللوحة الأولى لوحدها، ثم مساحة اللوحة الثانية لوحدها ونجمعهما. ولكن، إذا كانت اللوحتان متقاطعتين، فهذا يعني أننا حسبنا المساحة المشتركة مرتين!

لذا يجب علينا إيجاد إحداثيات مستطيل التقاطع باستخدام max لنقاط البداية و min لنقاط النهاية. إذا كان هناك تقاطع فعلي، نحسب مساحته ونطرحها من المجموع الكلي:

المساحة_الكلية = مساحة₁ + مساحة₂ - مساحة_التقاطع
📄 كود الحل
C++
#include <iostream>
#include <algorithm>
using namespace std;

int area(int x1, int y1, int x2, int y2) {
    return (x2 - x1) * (y2 - y1);
}

int main() {
    int x1 = 0, y1 = 0, x2 = 5, y2 = 5; // اللوحة الأولى
    int x3 = 3, y3 = 3, x4 = 8, y4 = 8; // اللوحة الثانية

    int total_area = area(x1, y1, x2, y2) + area(x3, y3, x4, y4);

    // إيجاد مستطيل التقاطع
    int cx1 = max(x1, x3), cy1 = max(y1, y3);
    int cx2 = min(x2, x4), cy2 = min(y2, y4);

    int intersect_area = 0;
    // التحقق من وجود تقاطع فعلي
    if (cx1 < cx2 && cy1 < cy2) {
        intersect_area = area(cx1, cy1, cx2, cy2);
    }

    cout << "المساحة الإجمالية المغطاة: " << total_area - intersect_area << endl;
    return 0;
}
02
نظام الصلاحيات السري
🟢 دمج Structs + Bit Manipulation
في مدرسة "نسمة"، يمتلك كل طالب ملفاً إلكترونياً. لتوفير مساحة الذاكرة، قرر المبرمج استخدام (رقم صحيح واحد) لتخزين جميع صلاحيات الطالب (قراءة، كتابة، وحذف) باستخدام الأقنعة الثنائية (Bitmasks).

البت رقم 0 = قراءة، البت رقم 1 = كتابة، البت رقم 2 = حذف.

المطلوب: قم بإنشاء هيكل (Struct) يمثل الطالب، وقم بكتابة كود يمنح طالباً جديداً صلاحية "الحذف" (بت رقم 2)، ثم يفحص ما إذا كان يمتلك صلاحية "الكتابة" (بت رقم 1) ويطبع النتيجة.
💡 الشرح المنطقي
نستخدم هيكل (Struct) اسمه Student لتجميع اسم الطالب مع رقم الصلاحيات الخاص به.

▶ لمنح صلاحية جديدة (تشغيل البت)، نستخدم بوابة أو (OR):
permissions |= (1 << 2)

▶ ولفحص ما إذا كانت صلاحية معينة موجودة، نستخدم بوابة و (AND):
permissions & (1 << 1)
📄 كود الحل
C++
#include <iostream>
#include <string>
using namespace std;

struct Student {
    string name;
    int permissions = 0; // 000 بالثنائي (لا يملك أي صلاحية)
};

int main() {
    Student ali;
    ali.name = "علي";

    // منح صلاحية الحذف (البت رقم 2)
    ali.permissions |= (1 << 2);

    cout << "صلاحيات علي الحالية بالرقم العشري: " << ali.permissions << endl;

    // فحص صلاحية الكتابة (البت رقم 1)
    if ((ali.permissions & (1 << 1)) != 0) {
        cout << ali.name << " يمتلك صلاحية الكتابة." << endl;
    } else {
        cout << ali.name << " لا يمتلك صلاحية الكتابة!" << endl;
    }

    return 0;
}
03
تحدي قفز الحواجز
🟠 البرمجة الديناميكية 1D · Bottom-Up
تشارك في سباق مدرسي لقفز الحواجز. هناك N من الحواجز المرقمة من 0 إلى N-1. عند كل حاجز i، تدفع كلفة من طاقتك للوقوف عليه قدرها cost[i].

يمكنك القفز إما للحاجز التالي (i+1) أو تخطي حاجز والقفز للذي يليه (i+2).

أوجد أقل كمية من الطاقة (كلفة) يمكنك صرفها للوصول إلى خط النهاية (تجاوز جميع الحواجز).
💡 الشرح المنطقي
هذه مسألة برمجة ديناميكية أحادية البعد. مصفوفة dp[i] ستمثل "أقل طاقة مطلوبة للوصول من الحاجز i إلى النهاية".

بأسلوب (من أسفل لأعلى Bottom-Up)، نبدأ من خط النهاية ونعود للخلف.

▶ المعادلة السحرية: طاقتك الحالية = كلفة وقوفك + أقل طاقة ممكنة من المسارين:
dp[i] = cost[i] + min(dp[i+1], dp[i+2])
📄 كود الحل
C++
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    vector<int> cost = {10, 15, 20}; // كلفة الوقوف على كل حاجز
    int n = cost.size();
    
    // مصفوفة DP مع خانات إضافية بصفر لتجنب الخروج عن الحدود
    vector<int> dp(n + 2, 0); 
    
    for (int i = n - 1; i >= 0; i--) {
        dp[i] = cost[i] + min(dp[i+1], dp[i+2]);
    }

    // يمكننا بدء السباق إما من الحاجز 0 أو الحاجز 1
    int min_energy = min(dp[0], dp[1]);
    
    cout << "أقل طاقة مبذولة لإنهاء السباق هي: " << min_energy << endl;
    return 0;
}

حقيبة ألعاب الفيديو
🟣 البرمجة الديناميكية 2D · 0/1 Knapsack
حصلت على حقيبة هدايا تتسع لوزن أقصاه W كيلوجرام. أمامك 3 أجهزة ألعاب (PlayStation, Xbox, Nintendo)، كل جهاز له "وزن" و "مستوى متعة".

المطلوب: اختر الأجهزة التي ستضعها في حقيبتك لتحصل على "أقصى متعة ممكنة" دون أن يتجاوز الوزن W.

⚠️ لا يمكنك أخذ نصف جهاز، إما أن تأخذه كله أو تتركه.
💡 الشرح المنطقي
هذه هي مسألة "حقيبة الظهر" الشهيرة (0/1 Knapsack). نحلها بمصفوفة 2D حيث:
dp[i][w] = أقصى متعة من أول i أجهزة إذا كان متبقياً وزن w.

في كل خطوة، إذا كان الجهاز أثقل من الوزن المتبقي، نتركه إجبارياً.
إذا كان يمكن حمله، نقارن بين خيارين لنأخذ الأفضل:

1️⃣ نتركه → dp[i-1][w]
2️⃣ نأخذه → value[i] + dp[i-1][w - weight[i]]
📄 كود الحل
C++
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int W = 5; // سعة الحقيبة
    vector<int> weight = {0, 2, 3, 4}; // أوزان الألعاب (بدأنا بـ 0 لسهولة المؤشرات)
    vector<int> value  = {0, 3, 4, 5}; // مستوى المتعة لكل لعبة
    int n = 3; // عدد الألعاب

    // مصفوفة 2D بأصفار
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));

    for (int i = 1; i <= n; i++) {
        for (int w = 0; w <= W; w++) {
            if (weight[i] > w) {
                dp[i][w] = dp[i-1][w]; // تركه إجبارياً
            } else {
                // الأفضل بين "تركه" و "أخذه"
                dp[i][w] = max(dp[i-1][w], value[i] + dp[i-1][w - weight[i]]);
            }
        }
    }

    cout << "أقصى متعة ممكنة يمكنك الحصول عليها هي: " << dp[n][W] << endl;
    return 0;
}
