يُعطى لك عدد صحيح n. تقوم برمي نرد عادي بستة أوجه (القيم من 1 إلى 6) وتجمع النتائج.


مهمتك هي حساب عدد الطرق للحصول على مجموع يساوي بالضبط n عن طريق رمي النرد مرة أو أكثر.

1+1+1 1+2 2+1 3

القيود: 1 ≤ n ≤ 10⁶  |  اطبع الناتج mod 10⁹+7

🪜 خطوات الحل
1
تعريف الحالة dp[x]

نعرّف dp[x] = عدد الطرق للحصول على مجموع يساوي x باستخدام رميات النرد (مع اعتبار الترتيب).

2
الحالة الأساسية

dp[0] = 1 — يوجد طريقة واحدة فقط للحصول على مجموع صفر: التسلسل الفارغ (لا نرمي أبداً).

3
العلاقة العودية

للوصول إلى x، آخر رمية كانت k ∈ {1..6}، فالمجموع السابق كان x−k. إذاً نجمع جميع الطرق السابقة الممكنة.

4
ترتيب الحساب التصاعدي

نحسب dp[1], dp[2], ..., dp[n] بالترتيب. كل خلية تعتمد فقط على خلايا سابقة لها — لذا الترتيب التصاعدي صحيح دائماً.

5
أخذ باقي القسمة

بعد كل جمع نأخذ باقي القسمة على 10⁹+7 لأن القيم قد تكبر جداً.

🔁 العلاقة العودية
الصيغة الرياضية
dp[0] = 1 // الحالة الأساسية

dp[x] = Σ dp[x − k] for k = 1..6, x−k ≥ 0 // mod 10⁹+7
🔬 شرح قطع الكود الرئيسية
الإعداد والمتغيرات
const int MOD = 1'000'000'007 — ثابت المودولو لتجنّب overflow.
vector<int> dp(n+1, 0) — مصفوفة بحجم n+1 مهيّأة بأصفار. نستخدم int لأن القيم بعد المودولو أصغر من 10⁹ تناسب int.
dp[0] = 1 — الحالة الأساسية: طريقة واحدة للحصول على صفر. 
const int MOD = 1'000'000'007;

int n;
cin >> n;

vector<int> dp(n + 1, 0);
dp[0] = 1; // الحالة الأساسية: التسلسل الفارغ
الحلقة الرئيسية — حساب dp
الحلقة الخارجية x = 1..n — نحسب عدد الطرق للوصول لكل مجموع ممكن.
الحلقة الداخلية k = 1..6 — تمثّل الوجوه الستة للنرد. نتحقق أن x−k ≥ 0 قبل الإضافة.
long long ways — نجمع في long long لتجنّب overflow قبل أخذ المودولو. 
for (int x = 1; x <= n; x++) {
    long long ways = 0;
    for (int k = 1; k <= 6; k++) {
        if (x - k >= 0) {
            ways += dp[x - k]; // آخر رمية كانت k
        }
    }
    dp[x] = (int)(ways % MOD);
}

dice_combinations.cpp 
#include <iostream>
#include <vector>
using namespace std;

const int MOD = 1'000'000'007;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> dp(n + 1, 0);
    dp[0] = 1; // one way to make sum 0: empty sequence

    for (int x = 1; x <= n; x++) {
        long long ways = 0;
        for (int k = 1; k <= 6; k++) {
            if (x - k >= 0) {
                ways += dp[x - k];
                if (ways >= MOD) ways -= MOD; // small optimization
            }
        }
        dp[x] = (int)ways;
    }

    cout << dp[n] << '\n';
    return 0;
}
⚡ التعقيد
التعقيد الزمني
O(n × 6) = O(n)
التعقيد المكاني
O(n)
⚠️
انتبه: دائماً خذ المودولو على 10⁹+7 بعد كل جمع. استخدم long long أثناء الجمع لتجنّب تجاوز السعة، ثم حوّل النتيجة إلى int.

🪜 خطوات الحل
1
قراءة الشبكة وتعريف dp[i][j]

نقرأ الشبكة ونعرّف dp[i][j] = عدد المسارات للوصول إلى الخلية (i,j) من (0,0) بالتحرك يميناً أو لأسفل فقط.

2
الحالة الأساسية

dp[0][0] = 1 إذا لم تكن فخاً. أما الفخاخ فتُعطى القيمة صفر مباشرة.

3
العلاقة العودية

كل خلية (i,j) يمكن الوصول إليها فقط من أعلاها (i-1,j) أو من يسارها (i,j-1). نجمع قيمتيهما إذا كانت الخلية فارغة.

4
ملء الجدول صف بصف

نكرر على i من 0 إلى n-1، وداخل كل صف على j من 0 إلى n-1. هذا يضمن أن dp[i-1][j] و dp[i][j-1] محسوبتان مسبقاً.

5
الإجابة في dp[n-1][n-1]

نطبع قيمة الخلية السفلية اليمنى mod 10⁹+7.

🔁 العلاقة العودية
الصيغة الرياضية
dp[0][0] = 1 // إذا لم تكن فخًا

dp[i][j] = 0 // إذا grid[i][j] == '*'

dp[i][j] = dp[i−1][j] + dp[i][j−1] // mod 10⁹+7
// مع اعتبار أي مؤشر خارج الحدود = 0
🔬 شرح قطع الكود الرئيسية
قراءة الشبكة وتهيئة dp
vector<string> grid(n) — نقرأ كل صف كسلسلة نصية مباشرة.
vector<vector<long long>> dp — جدول DP ثنائي الأبعاد مهيّأ بأصفار. نستخدم long long لتجنّب overflow قبل المودولو. 
vector<string> grid(n);
for (int i = 0; i < n; ++i)
    cin >> grid[i];

// جدول dp مهيّأ بأصفار
vector<vector<long long>> dp(n, vector<long long>(n, 0));
ملء الجدول — قلب الخوارزمية
فحص الفخ أولاً — إذا كانت الخلية فخاً نضع صفراً ونتابع.
الحالة الأساسية — الخلية (0,0) لها مسار واحد (نقطة البداية).
from_up و from_left — نتحقق من الحدود قبل الإضافة لتجنّب الخروج عن المصفوفة. 
for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
        if (grid[i][j] == '*') {
            dp[i][j] = 0; // فخ — لا نمر
            continue;
        }
        if (i == 0 && j == 0) {
            dp[i][j] = 1; // نقطة البداية
        } else {
            long long from_up   = (i > 0 ? dp[i-1][j] : 0);
            long long from_left = (j > 0 ? dp[i][j-1] : 0);
            dp[i][j] = (from_up + from_left) % MOD;
        }
    }
}

grid_paths.cpp 
#include <iostream>
#include <vector>
using namespace std;

const long long MOD = 1000000007LL;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<string> grid(n);
    for (int i = 0; i < n; ++i)
        cin >> grid[i];

    // dp[i][j] = number of ways to reach (i, j)
    vector<vector<long long>> dp(n, vector<long long>(n, 0));

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (grid[i][j] == '*') {
                dp[i][j] = 0; // cannot use traps
                continue;
            }
            if (i == 0 && j == 0) {
                dp[i][j] = 1; // starting cell
            } else {
                long long from_up   = (i > 0 ? dp[i - 1][j] : 0);
                long long from_left = (j > 0 ? dp[i][j - 1] : 0);
                dp[i][j] = (from_up + from_left) % MOD;
            }
        }
    }

    cout << dp[n - 1][n - 1] << "\n";
    return 0;
}
⚡ التعقيد
التعقيد الزمني
O(n²)
التعقيد المكاني
O(n²)
⚠️
انتبه: تحقق دائماً من الحدود باستخدام i > 0 و j > 0 قبل الوصول إلى dp[i-1][j] أو dp[i][j-1]. إذا كانت خلية البداية (0,0) فخاً، فالإجابة صفر تلقائياً.
