Frog 1

AtCoder Easy · سهل
📋
نص المسألة

لديك مصفوفة تحتوي على N عنصرًا: a[0], a[1], …, a[N−1]. تبدأ من الموقع 0 وتريد الوصول إلى الموقع N−1. في كل خطوة يمكنك:

التحرك من الموقع i إلى i+1 بتكلفة |a[i+1] − a[i]|
التحرك من الموقع i إلى i+2 بتكلفة |a[i+2] − a[i]|

احسب أقل تكلفة إجمالية للانتقال من الموقع 0 إلى الموقع N−1.

💡
الحل بالبرمجة الديناميكية

نعرّف الحالة dp[i] = أقل تكلفة للوصول إلى الموقع i من الموقع 0.

dp[0] = 0
الحالة الأساسية: نحن بالفعل في الموقع 0
dp[i] = min(dp[i−1] + |a[i]−a[i−1]|, dp[i−2] + |a[i]−a[i−2]|)
نصل إلى i إما من i−1 أو من i−2، ونختار الأرخص
تعقيد الزمن
O(N)
تعقيد المساحة
O(N)

الرسم التفاعلي — تتبع خوارزمية DP

الكود (C++)
C++ 
// مصفوفة الإدخال والـ DP
#include <iostream>
using namespace std;

int a[200000];
int dp[200000];

int f(int i) {
    // الحالات الأساسية
    if (i == 0) return 0;
    if (i == 1) return abs(a[1] - a[0]);

    // إذا تم حساب الحل من قبل
    if (dp[i] != -1) return dp[i];

    dp[i] = min(
        f(i - 1) + abs(a[i] - a[i - 1]),
        f(i - 2) + abs(a[i] - a[i - 2])
    );
    return dp[i];
}

int main() {
    // تهيئة dp بـ -1
    for (int i = 0; i < 200000; i++) dp[i] = -1;

    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    cout << f(n - 1) << "\n";
}

Vacation

AtCoder Easy · سهل
📋
نص المسألة

لديك ثلاث مصفوفات A وB وC، كل منها بطول N. في كل يوم i (من 0 إلى N−1) يجب اختيار نشاط واحد من A[i] أو B[i] أو C[i].

⚠️ الشرط: النشاط في اليوم i يجب أن يختلف عن النشاط في اليوم i+1

أوجد أكبر مجموع ممكن للقيم المختارة.

💡
الحل بالبرمجة الديناميكية

نعرّف dp[i][j] = أكبر مجموع للأيام 0..i إذا اخترنا النشاط j في اليوم i.

dp[0][0] = A[0]  |  dp[0][1] = B[0]  |  dp[0][2] = C[0]
الحالة الأساسية: اليوم الأول
dp[i][0] = max(dp[i−1][1], dp[i−1][2]) + A[i]
نشاط A لا يتبع نشاط A
dp[i][1] = max(dp[i−1][0], dp[i−1][2]) + B[i]
نشاط B لا يتبع نشاط B
dp[i][2] = max(dp[i−1][0], dp[i−1][1]) + C[i]
نشاط C لا يتبع نشاط C
تعقيد الزمن
O(N)
تعقيد المساحة
O(N)

الكود (C++)
C++ 
#include <iostream>
using namespace std;

int dp[200000][3];
int d[200000][3];

int f(int i, int type) {
    if (i == 0) return d[i][type];
    if (dp[i][type] != -1) return dp[i][type];

    if (type == 0)
        dp[i][type] = max(f(i-1,1), f(i-1,2)) + d[i][type];
    else if (type == 1)
        dp[i][type] = max(f(i-1,0), f(i-1,2)) + d[i][type];
    else
        dp[i][type] = max(f(i-1,0), f(i-1,1)) + d[i][type];

    return dp[i][type];
}

int main() {
    for (int i=0; i<200000; i++)
        for (int j=0; j<3; j++) dp[i][j] = -1;

    int n; cin >> n;
    for (int i=0; i<n; i++)
        for (int j=0; j<3; j++) cin >> d[i][j];

    cout << max({f(n-1,0), f(n-1,1), f(n-1,2)}) << "\n";
}

Staircase Sum — مجموع الدرج

مسألة جديدة Easy · سهل
📋
نص المسألة

لديك درج يحتوي على N درجة. كل درجة i لها قيمة v[i]. تبدأ من خارج الدرج (قبل الدرجة 0) وتريد الوصول إلى الدرجة N−1. في كل خطوة يمكنك القفز إلى الدرجة التالية أو تخطي درجة واحدة (القفز بمقدار درجتين).

القفز من الموقع i إلى i+1 — تجمع قيمة الدرجة i+1
القفز من الموقع i إلى i+2 — تجمع قيمة الدرجة i+2

احسب أكبر مجموع ممكن للقيم التي تجمعها حين تصل إلى الدرجة N−1.
ملاحظة: هذه المسألة مشابهة لـ Frog 1 لكن بدلاً من تقليل التكلفة نريد تعظيم المجموع.

💡
الحل بالبرمجة الديناميكية

نعرّف dp[i] = أكبر مجموع للقيم إذا وصلنا إلى الدرجة i.

dp[0] = v[0]
الحالة الأساسية: نبدأ بجمع قيمة الدرجة الأولى
dp[i] = max(dp[i−1], dp[i−2]) + v[i]
نصل إلى i إما من i−1 أو من i−2، ونختار الأعلى

الإجابة النهائية هي dp[N−1].

تعقيد الزمن
O(N)
تعقيد المساحة
O(N)
🎮
الرسم التفاعلي — درج القفز
السرعة
اضغط "تشغيل" لبدء الرسم المتحرك...
💻
الكود (C++)
C++ 
#include <iostream>
using namespace std;

int v[200000];
int dp[200000];

int f(int i) {
    // الحالات الأساسية
    if (i == 0) return v[0];
    if (i == 1) return max(v[0], v[0]) + v[1];

    if (dp[i] != -1) return dp[i];

    dp[i] = max(f(i-1), f(i-2)) + v[i];
    return dp[i];
}

int main() {
    for (int i=0; i<200000; i++) dp[i] = -1;

    int n; cin >> n;
    for (int i=0; i<n; i++) cin >> v[i];

    cout << f(n-1) << "\n";
}
📝
مثال توضيحي

للمصفوفة: v = [3, 1, 5, 2, 4] (5 درجات، الفهرسة من 0)

dp[0] = 3
dp[1] = max(dp[0], ?) + v[1] = max(3, نبدأ من الخارج=0) + 1 = 3 + 1 = 4
الوصول إلى الدرجة 1 إما من 0 (dp=3) أو من الخارج قفزتين (=0)
dp[2] = max(dp[1], dp[0]) + v[2] = max(4, 3) + 5 = 9
dp[3] = max(dp[2], dp[1]) + v[3] = max(9, 4) + 2 = 11
dp[4] = max(dp[3], dp[2]) + v[4] = max(11, 9) + 4 = 15
أكبر مجموع = 15

المسار الأمثل: 0 → 2 → 4 بمجموع 3 + 5 + 4 = 12 أو: 0 → 1 → 2 → 3 → 4 = 3+1+5+2+4 = 15 ✓
