1. المسألة المطلوبة

اكتب برنامجاً بلغة C++ يستخدم الهياكل (Structs) والمصفوفات (Arrays) لإدارة بيانات الطلاب. يجب أن يقوم البرنامج بما يلي:

  • سؤال المستخدم عن عدد الطلاب المراد إدخال بياناتهم.
  • استخدام حلقة تكرار (For Loop) لاستقبال بيانات كل طالب (الرقم الجامعي، الاسم، درجات 3 مواد).
  • استخدام حلقة تكرار أخرى لحساب المعدل وطباعة النتيجة النهائية لكل طالب.

2. الحل (كود C++)

#include <iostream>
#include <string>
using namespace std;

// 1. تعريف البنية
struct Student {
    int id;
    string name;
    double mathScore;
    double scienceScore;
    double progScore;
};

int main() {
    int count;
    cout << "How many students? ";
    cin >> count;

    // 2. إنشاء المصفوفة
    Student students[50]; 

    // 3. حلقة الإدخال
    for (int i = 0; i < count; i++) {
        cout << "\n--- Student " << i + 1 << " ---" << endl;
        cout << "ID: "; cin >> students[i].id;
        cout << "Name: "; cin >> students[i].name;
        cout << "Math: "; cin >> students[i].mathScore;
        cout << "Science: "; cin >> students[i].scienceScore;
        cout << "Programming: "; cin >> students[i].progScore;
    }

    // 4. حلقة الإخراج والنتائج
    cout << "\n=== Results ===" << endl;
    for (int i = 0; i < count; i++) {
        double total = students[i].mathScore + students[i].scienceScore + students[i].progScore;
        double average = total / 3.0;
        
        cout << students[i].name << " (ID: " << students[i].id << ") - Average: " << average << endl;
    }

    return 0;
}
