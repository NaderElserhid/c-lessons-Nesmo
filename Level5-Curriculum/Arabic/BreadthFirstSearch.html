تموّجات الماء وانتشار الخبر

ألقِ حصاة في بركة ماء: تنتشر الدوائر موجة بعد موجة، فتصل الموجة الأولى إلى كل ما يبعد خطوة واحدة، ثم الثانية إلى ما يبعد خطوتين، وهكذا. وكذلك انتشار خبر في الفصل: في الدقيقة الأولى يعرفه أصدقاؤك المباشرون، وفي الثانية أصدقاء أصدقائك... هذه هي فكرة البحث بالعرض أولًا (Breadth First Search أو BFS): زُر كل ما يبعد خطوة، ثم كل ما يبعد خطوتين.

والنتيجة المهمة: في رسم غير مرجَّح، BFS يعطينا أقصر مسار (بأقل عدد حواف) من رأس البداية إلى كل رأس آخر.

BFS بالطابور

نستعمل طابورًا (queue): نضع الرؤوس في نهايته ونأخذ من مقدمته. بهذا يخرج القريب (الذي دخل قبل) قبل البعيد. ونستعمل المصفوفة dist التي تحمل المسافة من البداية، والقيمة -1 تعني «لم يُزر»:

bfs.cpp
#include <iostream>#include <vector>#include <queue>using namespace std; int main() {    int n = 8;    int edges[8][2] = {{1, 2}, {1, 3}, {2, 4}, {2, 5}, {3, 6}, {3, 7}, {5, 8}, {6, 8}};    vector<vector<int>> adj(n + 1);    for (auto& e : edges) {        adj[e[0]].push_back(e[1]);        adj[e[1]].push_back(e[0]);    }     vector<int> dist(n + 1, -1);    queue<int> q;    dist[1] = 0;    q.push(1);     cout << "order:";    while (!q.empty()) {        int v = q.front();        q.pop();        cout << " " << v;        for (int u : adj[v]) {            if (dist[u] == -1) {                dist[u] = dist[v] + 1;                q.push(u);            }        }    }    cout << endl;     for (int v = 1; v <= n; v++) {        cout << "distance to " << v << " = " << dist[v] << endl;    }    return 0;}

ترتيب الخروج 1 2 3 4 5 6 7 8 يتبع المستويات: المستوى 0 هو 1، والمستوى 1 هو 2 و3، والمستوى 2 هو 4 و5 و6 و7، والمستوى 3 هو 8. لاحظ أن المسافة إلى 8 هي 3 وليست 4 رغم أن له جارين، لأن BFS يصل إليه أولًا عبر 5.

اكتب dist[u] (أو علّم u مزورًا) عند وضعه في الطابور وليس عند إخراجه. لو انتظرت الإخراج لدخل الرأس نفسه الطابور مرات كثيرة عبر جيران مختلفين، وتضيع كفاءة الخوارزمية.

مختبر تفاعلي (1): BFS خطوة بخطوة

اختر رسمًا ورأس البداية. في كل ضغطة: إما أن نُخرج رأسًا من مقدمة الطابور (برتقالي)، أو نضيف جارًا جديدًا إلى نهايته (برتقالي فاتح). الرقم تحت الرأس هو مسافته. الخط الأزرق العريض يوصل الرأس بالذي اكتشفه.

المتاهات: BFS على الشبكة

المتاهة شبكة: كل خلية رأس، وجيرانها الأربعة (إن لم تكن جدرانًا) متصلة بها. أقصر طريق من S إلى E هو تطبيق مباشر لـ BFS. نستعمل مصفوفتي الاتجاهات لنتجنب تكرار الكود:

maze_bfs.cpp
#include <iostream>#include <vector>#include <queue>#include <string>using namespace std; int main() {    vector<string> g = {        "S..#....",        ".#.#.##.",        ".#...#..",        ".####.#.",        "......#E"    };    int R = 5, C = 8;    int dr[4] = {1, -1, 0, 0};    int dc[4] = {0, 0, 1, -1};     vector<vector<int>> dist(R, vector<int>(C, -1));    vector<vector<pair<int, int>>> from(R, vector<pair<int, int>>(C, {-1, -1}));    queue<pair<int, int>> q;    dist[0][0] = 0;    q.push({0, 0});    while (!q.empty()) {        auto [r, c] = q.front();        q.pop();        for (int k = 0; k < 4; k++) {            int nr = r + dr[k], nc = c + dc[k];            if (nr < 0 || nr >= R || nc < 0 || nc >= C) {                continue;            }            if (g[nr][nc] == '#' || dist[nr][nc] != -1) {                continue;            }            dist[nr][nc] = dist[r][c] + 1;            from[nr][nc] = {r, c};            q.push({nr, nc});        }    }     cout << "steps to E: " << dist[4][7] << endl;    int r = 4, c = 7;    while (from[r][c].first != -1) {        auto p = from[r][c];        r = p.first;        c = p.second;        if (g[r][c] == '.') {            g[r][c] = '*';        }    }    for (auto& row : g) {        cout << row << endl;    }    return 0;}

مختبر تفاعلي (2): موجة المتاهة

انقر على الخلايا لوضع الجدران أو إزالتها (اختر «جدار»)، أو اختر «بداية» أو «نهاية» ثم انقر لتحريكهما. اضغط «الموجة التالية» لترى BFS ينتشر مستوى بعد مستوى، والأرقام هي المسافات. عندما تصل الموجة إلى النهاية يُرسم المسار الأقصر بالأخضر.

مشروع الدرس: عدّاء المتاهة

نحسب أقل عدد خطوات للخروج من متاهة.

الفكرة

في لعبة، يقف اللاعب في الخلية S ويريد الوصول إلى الخلية E. يتحرك خطوة واحدة في كل مرة إلى خلية مجاورة (فوق أو تحت أو يمين أو يسار)، ولا يستطيع دخول الجدران #. المطلوب أقل عدد خطوات، أو أن اللاعب محبوس.

المطلوب منك
  1. اقرأ R وC ثم R سطرًا، وحدّد موقعي S وE أثناء القراءة.
  2. شغّل BFS من S بمصفوفة مسافات dist (قيمتها -1 لما لم يُزر).
  3. إذا كانت dist للنهاية -1 فاطبع No path وإلا اطبع Steps: ثم المسافة.
مثالان على التشغيل

التشغيل الأول (المتاهة التي رأيناها):

5 8 S..#.... .#.#.##. .#...#.. .####.#. ......#E
Steps: 15

التشغيل الثاني (جدار يسد الطريق):

2 3 S#E .#.
No path

تحقق ذهنيًا: في الثاني، العمود الأوسط كله جدران، فلا يستطيع اللاعب العبور من اليسار إلى اليمين مهما التف. وفي الأول المسار يمر بـ 15 خلية بعد البداية.

بعد أن تجرّب بنفسك: اعرض الحل
maze_runner.cpp
#include <iostream>
#include <vector>
#include <queue>
#include <string>
using namespace std;

int main() {
    int R, C;
    cin >> R >> C;
    vector<string> g(R);
    int sr = 0, sc = 0, er = 0, ec = 0;
    for (int r = 0; r < R; r++) {
        cin >> g[r];
        for (int c = 0; c < C; c++) {
            if (g[r][c] == 'S') {
                sr = r;
                sc = c;
            }
            if (g[r][c] == 'E') {
                er = r;
                ec = c;
            }
        }
    }

    int dr[4] = {1, -1, 0, 0};
    int dc[4] = {0, 0, 1, -1};
    vector<vector<int>> dist(R, vector<int>(C, -1));
    queue<pair<int, int>> q;
    dist[sr][sc] = 0;
    q.push({sr, sc});
    while (!q.empty()) {
        auto [r, c] = q.front();
        q.pop();
        for (int k = 0; k < 4; k++) {
            int nr = r + dr[k], nc = c + dc[k];
            if (nr < 0 || nr >= R || nc < 0 || nc >= C) {
                continue;
            }
            if (g[nr][nc] == '#' || dist[nr][nc] != -1) {
                continue;
            }
            dist[nr][nc] = dist[r][c] + 1;
            q.push({nr, nc});
        }
    }

    if (dist[er][ec] == -1) {
        cout << "No path" << endl;
    } else {
        cout << "Steps: " << dist[er][ec] << endl;
    }
    return 0;
}

1

أقل عدد محطات

متوسط

مترو فيه n محطة وm خط توصيل بين محطتين (في الاتجاهين). اقرأ s وt واطبع أقل عدد خطوط يلزم لركوبها من المحطة s إلى t، أو -1 إذا لم يمكن الوصول.

المطلوب
  • ابنِ قائمة الجوار.
  • شغّل BFS من s مع dist.
  • اطبع dist[t].
مثال على الإدخال والناتج المتوقع
8 8 1 8 1 2 1 3 2 4 2 5 3 6 3 7 5 8 6 8
3
تلميحات
  • القيمة -1 الأصلية للمسافة تصلح نتيجةً عندما لا يوجد مسار.
  • في المثال (الرسم 1) الجواب من 1 إلى 8 هو 3.
نموذج الحل (بعد المحاولة)
fewest_stops.cpp
// Challenge 1: the fewest stops between two stations
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    int n, m, s, t;
    cin >> n >> m >> s >> t;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<int> dist(n + 1, -1);
    queue<int> q;
    dist[s] = 0;
    q.push(s);
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int u : adj[v]) {
            if (dist[u] == -1) {
                dist[u] = dist[v] + 1;
                q.push(u);
            }
        }
    }
    cout << dist[t] << endl;
    return 0;
}
2

فريقان

متوسط

في الفصل n طالبًا، وبعض الأزواج لا يحبون اللعب في الفريق نفسه (أعداء). هل يمكن تقسيم الطلاب إلى فريقين بحيث لا يجتمع عدوان في فريق واحد؟ اطبع YES أو NO.

المطلوب
  • اعتبر العداوة حافة بين طالبين.
  • شغّل BFS من كل رأس لم يُلوَّن، وأعطِ الرأس البداية اللون 0.
  • كل جار يأخذ اللون المعاكس.
  • إذا التقيت جارًا له اللون نفسه فالتقسيم مستحيل.
مثال على الإدخال والناتج المتوقع
5 5 1 2 2 3 3 4 4 5 5 1
NO
تلميحات
  • حلقة فردية الطول (مثل 5 طلاب في دائرة) تجعل التقسيم مستحيلًا.
  • في المثال دائرة من 5 طلاب، فالجواب NO.
نموذج الحل (بعد المحاولة)
two_teams.cpp
// Challenge 2: can the students be split into two teams?
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<int> color(n + 1, -1);
    bool ok = true;
    for (int s = 1; s <= n; s++) {
        if (color[s] != -1) {
            continue;
        }
        color[s] = 0;
        queue<int> q;
        q.push(s);
        while (!q.empty()) {
            int v = q.front();
            q.pop();
            for (int u : adj[v]) {
                if (color[u] == -1) {
                    color[u] = 1 - color[v];
                    q.push(u);
                } else if (color[u] == color[v]) {
                    ok = false;
                }
            }
        }
    }
    cout << (ok ? "YES" : "NO") << endl;
    return 0;
}

ملخص سريع

  • BFS يزور الرؤوس مستوى بعد مستوى باستعمال **طابور**.
  • في رسم غير مرجَّح يعطي BFS أقصر مسار (بأقل عدد حواف) من البداية إلى كل رأس.
  • dist[v] = -1 تعني «لم يُزر»، ونكتب المسافة عند وضع الرأس في الطابور لا عند إخراجه.
  • الزمن O(n + m)، والذاكرة O(n).
  • لاسترجاع المسار نسجل parent[u] ثم نمشي من الهدف إلى الخلف ونعكس.
  • في الشبكات نستعمل مصفوفتي الاتجاهات dr وdc والخلية رأس.
  • نعالج مستوى كاملًا في كل جولة بأخذ q.size() في بدايتها.
