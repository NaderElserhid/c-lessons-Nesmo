البرمجة الديناميكية على الأشجار

Tree DP — كل رأس يحسب نتيجته من نتائج أبنائه

ماذا ستتعلم في هذا الدرس؟

  • تفهم فكرة حالة الرأس وعلاقتها بالأبناء وترتيب ما بعد الزيارة
  • تحسب مجموع الشجرة الفرعية بالبرمجة الديناميكية
  • تستعمل حالتين take وskip لحل المجموعة المستقلة العظمى
  • تحسب أطول مسار نازل وقطر الشجرة بمسح واحد
  • تحدد في أي مسألة: معنى الحالة، والحالة الأساسية، والعلاقة

كل مدير يسأل مرؤوسيه

مدير عام يريد معرفة إجمالي ما يحققه فرع شركته. لا يزور كل الموظفين بنفسه: يسأل مديري الأقسام، وكل مدير قسم يسأل رؤساء فرقه، وهكذا حتى الأوراق. كل واحد يجمع إجابات من تحته ثم يعطي إجابته لمن فوقه. هذه هي البرمجة الديناميكية على الأشجار (Tree DP).

الفكرة نفسها التي رأيناها في البرمجة الديناميكية: نعرّف حالة (هنا: «ماذا نعرف عن الشجرة الفرعية لرأس v؟») ونكتب علاقة تحسبها من حالات الأبناء. ولأن الأبناء يُحسبون قبل الأب، يكون ترتيب الحساب هو ما بعد الزيارة (post-order) في DFS: نحل الأبناء أولًا ثم نجمع النتائج في الأب.

ثلاث خطوات تتكرر في كل مسألة: (1) ما معنى dp[v]؟ (2) ما قيمته للورقة (الحالة الأساسية)؟ (3) كيف أحسبه من dp أبنائي؟

أبسط مثال: مجموع الشجرة الفرعية

نعطي كل رأس قيمة (ذهب في منطقة مثلًا) ونريد لكل رأس مجموع قيم شجرته الفرعية. المعنى: sum[v] هو مجموع قيم v وأحفاده. الحالة الأساسية: قيمة v نفسه. العلاقة: نضيف sum[u] لكل ابن u:

subtree_sum.cpp
#include <iostream>#include <vector>#include <functional>using namespace std; int main() {    int n = 9;    int edges[8][2] = {{1, 2}, {1, 3}, {1, 4}, {2, 5}, {2, 6}, {4, 7}, {7, 8}, {7, 9}};    int val[10] = {0, 5, 3, 8, 4, 2, 6, 7, 1, 9};    vector<vector<int>> adj(n + 1);    for (auto& e : edges) {        adj[e[0]].push_back(e[1]);        adj[e[1]].push_back(e[0]);    }     vector<long long> sum(n + 1, 0);    function<void(int, int)> dfs = [&](int v, int p) {        sum[v] = val[v];        for (int u : adj[v]) {            if (u != p) {                dfs(u, v);                sum[v] += sum[u];            }        }    };    dfs(1, 0);     for (int v = 1; v <= n; v++) {        cout << "sum[" << v << "] = " << sum[v] << endl;    }    return 0;}

مختبر تفاعلي (1): ملء جدول الشجرة

اختر شجرة ومسألة، ثم اضغط «الخطوة التالية». نحسب رأسًا واحدًا في كل ضغطة، بترتيب ما بعد الزيارة (الأوراق أولًا والجذر آخرًا). الصيغة المستعملة تظهر تحت الرسم، والرقم تحت الرأس هو نتيجته.

حالتان لكل رأس: المجموعة المستقلة

مجموعة رؤوس مستقلة (independent set) هي التي لا يجمع بين أي اثنين منها حافة. نريد أكبر مجموعة. القرار عند الرأس v يؤثر على أبنائه: إن أخذناه فلا نأخذ أي ابن له. لذلك نحتاج حالتين:

الحالةالمعنىالعلاقة
take[v]أكبر مجموعة في شجرة v تضم v1 + مجموع skip[u] للأبناء
skip[v]أكبر مجموعة في شجرة v لا تضمهمجموع max(take[u], skip[u])

مختبر تفاعلي (2): تحدَّ البرمجة الديناميكية

اكتب أرقام الرؤوس التي تختارها مفصولة بمسافات. سنتحقق أنه لا يتجاور اثنان، ونخبرك كم أنت قريب من الأمثل. اضغط «اعرض مجموعة مثلى» لتحصل عليها بالبرمجة الديناميكية. الرؤوس المختارة بالأزرق، والحواف المخالفة بالأحمر.

القطر بالبرمجة الديناميكية

رأينا القطر بمسحين. وبالبرمجة الديناميكية نجده بمسح واحد: نعرّف down[v] أطول مسار نازل من v. أطول مسار في الشجرة له أعلى رأس فيه؛ ومن ذلك الرأس ينزل في فرعين (أو فرع واحد)، فطوله best1 + best2 حيث هما أكبر قيمتين من down[u] + 1 بين الأبناء:

int n = 9;int edges[8][2] = {{1, 2}, {1, 3}, {1, 4}, {2, 5}, {2, 6}, {4, 7}, {7, 8}, {7, 9}};vector<vector<int>> adj(n + 1);for (auto& e : edges) {    adj[e[0]].push_back(e[1]);    adj[e[1]].push_back(e[0]);}vector<int> down(n + 1, 0);int diameter = 0;function<void(int, int)> dfs = [&](int v, int p) {    int best1 = 0, best2 = 0;    for (int u : adj[v]) {        if (u != p) {            dfs(u, v);            int d = down[u] + 1;            if (d > best1) {                best2 = best1;                best1 = d;            } else if (d > best2) {                best2 = d;            }        }    }    down[v] = best1;    diameter = max(diameter, best1 + best2);};dfs(1, 0); cout << "down:";for (int v = 1; v <= n; v++) {    cout << " " << down[v];}cout << endl;cout << "diameter = " << diameter << endl;

مشروع الدرس: حفلة الشركة

نختار المدعوين لنحصل على أكبر مرح ممكن.

الفكرة

شركة فيها n موظفًا، الموظف 1 هو المدير العام ولكل موظف آخر مدير مباشر. لكل موظف «درجة مرح» تضيفها حضورُه. لا يريد أحد أن يحضر مع مديره المباشر، فلا يُدعى موظف ومديره المباشر معًا. المطلوب أكبر مجموع مرح ممكن.

المطلوب منك
  1. اقرأ n ثم درجات المرح للموظفين 1 إلى n، ثم مدير كل موظف من 2 إلى n.
  2. احسب بـ DFS من الموظف 1 المصفوفتين take وskip: take[v] يبدأ بمرح v ويضيف skip[u] لكل ابن، وskip[v] يضيف max(take[u], skip[u]).
  3. اطبع Maximum fun: ثم الأكبر من take[1] وskip[1].
مثالان على التشغيل

التشغيل الأول (الشجرة التي رأيناها):

9 5 3 8 4 2 6 7 1 9 1 1 1 2 2 4 7 7
Maximum fun: 30

التشغيل الثاني (سلسلة من خمسة موظفين):

5 4 1 6 2 5 1 2 3 4
Maximum fun: 15

تحقق ذهنيًا: في السلسلة لا يصح دعوة اثنين متتاليين، فأفضل خيار هو الموظفون 1 و3 و5 بمرح 4 + 6 + 5 = 15.

بعد أن تجرّب بنفسك: اعرض الحل
company_party.cpp
#include <iostream>
#include <vector>
#include <functional>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> fun(n + 1);
    for (int v = 1; v <= n; v++) {
        cin >> fun[v];
    }
    vector<vector<int>> children(n + 1);
    for (int v = 2; v <= n; v++) {
        int boss;
        cin >> boss;
        children[boss].push_back(v);
    }

    vector<int> take(n + 1, 0), skip(n + 1, 0);
    function<void(int)> dfs = [&](int v) {
        take[v] = fun[v];
        for (int u : children[v]) {
            dfs(u);
            take[v] += skip[u];
            skip[v] += max(take[u], skip[u]);
        }
    };
    dfs(1);

    cout << "Maximum fun: " << max(take[1], skip[1]) << endl;
    return 0;
}

ذهب كل منطقة

سهل

مملكة فيها n منطقة، والمنطقة 1 هي العاصمة، ولكل منطقة أخرى منطقة أم. اقرأ كمية الذهب في كل منطقة ثم أم كل منطقة من 2 إلى n. اطبع لكل منطقة مجموع الذهب فيها وفي كل المناطق التي تتبعها.

المطلوب
  • خزّن قائمة الأبناء لكل منطقة.
  • بـ DFS من العاصمة احسب total[v] بعد العودة من الأبناء.
  • اطبع total للمناطق 1 إلى n مفصولة بمسافات.
مثال على الإدخال والناتج المتوقع
9 5 3 8 4 2 6 7 1 9 1 1 1 2 2 4 7 7
45 11 8 21 2 6 17 1 9
تلميحات
  • استعمل long long لأن المجاميع قد تكبر.
  • في المثال الجواب 45 11 8 21 2 6 17 1 9.

#include <vector>

القطر بالبرمجة الديناميكية

متوسط

اقرأ شجرة من n رأسًا وn - 1 حافة. اطبع قطرها (أطول مسار بين رأسين بعدد الحواف) دون استعمال BFS، بل بحساب down في DFS واحد.

المطلوب
  • عرّف down[v] أطول مسار نازل من v.
  • لكل رأس احتفظ بأكبر قيمتين best1 وbest2 من down[u] + 1 بين أبنائه.
  • حدّث القطر بـ best1 + best2 وأعد best1 قيمةً لـ down[v].
مثال على الإدخال والناتج المتوقع
6 1 2 1 3 3 4 3 5 5 6
4
تلميحات
  • الرأس ذو الابن الواحد يعطي best2 = 0، فالمسار ينتهي عنده.
  • في المثال الجواب 4.
نموذج الحل (بعد المحاولة)
dp_diameter.cpp
// Challenge 2: the diameter with dynamic programming
#include <iostream>
#include <vector>
#include <functional>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<int> down(n + 1, 0);
    int diameter = 0;
    function<void(int, int)> dfs = [&](int v, int p) {
        int best1 = 0, best2 = 0;
        for (int u : adj[v]) {
            if (u != p) {
                dfs(u, v);
                int d = down[u] + 1;
                if (d > best1) {
                    best2 = best1;
                    best1 = d;
                } else if (d > best2) {
                    best2 = d;
                }
            }
        }
        down[v] = best1;
        diameter = max(diameter, best1 + best2);
    };
    dfs(1, 0);
    cout << diameter << endl;
    return 0;
}

ملخص سريع

  • البرمجة الديناميكية على الأشجار: نعرّف حالة لكل رأس عن شجرته الفرعية ونحسبها من حالات أبنائه.
  • ترتيب الحساب هو ما بعد الزيارة: الأبناء أولًا ثم الأب.
  • نحدد ثلاثة أشياء: معنى dp[v]، وقيمته للورقة، وعلاقته بأبنائه.
  • مجموع الشجرة الفرعية: sum[v] = val[v] + مجموع sum[u].
  • للمجموعة المستقلة نحتاج حالتين: take[v] وskip[v]، والجواب max(take[1], skip[1]).
  • أطول مسار نازل down[v]، والقطر هو أكبر best1 + best2 على كل الرؤوس.
  • الزمن في هذه المسائل O(n) لأن كل رأس وكل حافة يُعالجان مرة.
