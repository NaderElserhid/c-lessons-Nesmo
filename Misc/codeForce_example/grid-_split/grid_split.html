أفضل شبكتين فرعيتين بدون تقاطع (2D Grid DP)

الفكرة الأساسية للطلاب:
نريد إيجاد شبكتين فرعيتين بحجم K x K بأكبر مجموع ممكن، ولكن بشرط ألا تتقاطعا.
كيف نضمن عدم التقاطع برمجياً؟ السر يكمن في رسم "خط فصل"!
إذا لم تتقاطع شبكتان، فهذا يعني حتماً أنه يمكننا رسم خط أفقي يفصل بينهما (واحدة فوق الخط والأخرى تحته)، أو خط عمودي (واحدة يمين الخط والأخرى يساره).
في الكود، نقوم بتجربة جميع خطوط الفصل الممكنة، ونستعين بمصفوفات التخزين المسبق (Prefix Maximums) لمعرفة أكبر شبكة في كل جهة بسرعة فائقة O(1).

1. خطوات الحل المنهجية

  1. بناء المصفوفة التراكمية (2D Prefix Sum): نبدأ بتجهيز مصفوفة تراكمية للشبكة الأساسية. هذا التكتيك ضروري لتسريع الحسابات، حيث يسمح لنا بإيجاد مجموع عناصر أي مربع فرعي بحجم K x K في خطوة واحدة O(1) بدلاً من المرور على عناصره المتعددة في كل مرة باستخدام حلقات تكرارية مكلفة للوقت.
  2. حساب مجاميع كل المربعات الممكنة: نمر على الشبكة باستخدام إحداثيات المصفوفة التراكمية، ونحسب مجموع كل مربع K x K محتمل داخل الشبكة، ثم نقوم بتخزين هذه النتائج الجاهزة في مصفوفة مساعدة (في كودنا أسميناها val).
  3. تجهيز مصفوفات القيم العظمى (Prefix Maximums): هنا يكمن جوهر البرمجة الديناميكية في المسألة. نقوم بإنشاء 4 مصفوفات منفصلة لتخزين "أكبر مجموع مربع متوفر" عند قطع الشبكة في اتجاه معين:
    • max_up: تخزن أكبر مجموع لمربع يقع بالكامل "أعلى" من صف معين.
    • max_down: تخزن أكبر مجموع لمربع يقع بالكامل "أسفل" من صف معين.
    • max_left: تخزن أكبر مجموع لمربع يقع بالكامل "يسار" عمود معين.
    • max_right: تخزن أكبر مجموع لمربع يقع بالكامل "يمين" عمود معين.
  4. تجربة خطوط الفصل وإيجاد الحل الأمثل: نمرر حلقة تكرارية لتجربة رسم "خط فصل وهمي" بين كل صفين، ونجمع (القيمة العظمى في الجزء العلوي + القيمة العظمى في الجزء السفلي). ثم نجرب نفس الشيء عمودياً بجمع (الجزء الأيمن + الجزء الأيسر). القيمة الأكبر التي تظهر من بين جميع محاولات الفصل هذه هي الحل النهائي.

3. الكود البرمجي (C++)

#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long INF = 2e18;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N, K;
    if (!(cin >> N >> K)) return 0;

    vector<vector<long long>> grid(N + 1, vector<long long>(N + 1, 0));
    vector<vector<long long>> pref(N + 1, vector<long long>(N + 1, 0));

    for (int i = 1; i <= N; ++i) {
        for (int j = 1; j <= N; ++j) {
            cin >> grid[i][j];
            pref[i][j] = grid[i][j] + pref[i-1][j] + pref[i][j-1] - pref[i-1][j-1];
        }
    }

    auto get_sum = [&](int r1, int c1, int r2, int c2) {
        return pref[r2][c2] - pref[r1-1][c2] - pref[r2][c1-1] + pref[r1-1][c1-1];
    };

    vector<vector<long long>> val(N + 1, vector<long long>(N + 1, -INF));
    for (int i = 1; i <= N - K + 1; ++i) {
        for (int j = 1; j <= N - K + 1; ++j) {
            val[i][j] = get_sum(i, j, i + K - 1, j + K - 1);
        }
    }

    vector<long long> max_up(N + 2, -INF);
    vector<long long> max_down(N + 2, -INF);
    vector<long long> max_left(N + 2, -INF);
    vector<long long> max_right(N + 2, -INF);

    for (int i = K; i <= N; ++i) {
        long long mx = -INF;
        int r = i - K + 1; 
        for (int c = 1; c <= N - K + 1; ++c) mx = max(mx, val[r][c]);
        max_up[i] = max(max_up[i-1], mx);
    }

    for (int i = N - K + 1; i >= 1; --i) {
        long long mx = -INF;
        int r = i; 
        for (int c = 1; c <= N - K + 1; ++c) mx = max(mx, val[r][c]);
        max_down[i] = max(max_down[i+1], mx);
    }

    for (int j = K; j <= N; ++j) {
        long long mx = -INF;
        int c = j - K + 1; 
        for (int r = 1; r <= N - K + 1; ++r) mx = max(mx, val[r][c]);
        max_left[j] = max(max_left[j-1], mx);
    }

    for (int j = N - K + 1; j >= 1; --j) {
        long long mx = -INF;
        int c = j; 
        for (int r = 1; r <= N - K + 1; ++r) mx = max(mx, val[r][c]);
        max_right[j] = max(max_right[j+1], mx);
    }

    long long ans = -INF;

    for (int i = K; i <= N - K; ++i) {
        if (max_up[i] != -INF && max_down[i+1] != -INF)
            ans = max(ans, max_up[i] + max_down[i+1]);
    }
    
    for (int j = K; j <= N - K; ++j) {
        if (max_left[j] != -INF && max_right[j+1] != -INF)
            ans = max(ans, max_left[j] + max_right[j+1]);
    }

    cout << ans << "\n";
    return 0;
}
