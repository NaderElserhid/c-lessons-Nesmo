حيل التعامل مع البِتّات (Bit Manipulation Tricks)

في الفصل السابق تعلّمت عمليات البِتّات الأساسية: & | ^ ~ << >>. الآن سنجمع هذه العمليات معًا لنبني حيلًا جاهزة وسريعة جدًا: كيف نفحص بِتًا واحدًا، كيف نغيّره، وكيف نستفيد من خاصية عجيبة في الطرح لإزالة أو عزل بِت معيّن دفعة واحدة.

الإعداد: المتغيّرات والأقنعة

سنفترض في كل هذا الفصل وجود متغيّرين:

setup.cpp
int x;       // عدد صحيح غير سالب
int k;       // رقم البِت (0-based)، حيث k=0 هو أقل بِت أهمية (LSB)

وسنعتمد بكثرة على القناع (mask) 1 << k، وهو عدد يكون فيه البِت رقم k فقط مساويًا لـ 1، وكل الباقي أصفار. كل الحيل في هذا الفصل هي ببساطة توليفات مختلفة من هذا القناع مع عمليات &، |، ^، ~.

فحص ما إذا كان البِت k يساوي 1

bool isSet = (x >> k) & 1;

  • x >> k → إزاحة لليمين بحيث ينتقل البِت رقم k إلى أقصى اليمين.

  • (x >> k) & 1 → تُبقي البِت الأيمن فقط وتُصفّر باقي البِتّات.

  • الناتج يكون 0 أو 1، وهو يعادل false أو true منطقيًا.

مثال

check_bit.cpp
int x = 10;     // 10 = 1010 (in binary)
int k = 1;

bool isSet = (x >> 1) & 1; // 1010 >> 1 = 101;  101 & 1 = 1 (true)
isSet = 1 (true)

ضبط البِت رقم k إلى 1

x |= (1 << k);

  • العامل | يعيّن البِت إلى 1 إذا كان أحد الطرفين على الأقل يحتوي 1.

  • بهذا نُجبر البِت رقم k على أن يصبح 1، بينما تبقى باقي البِتّات كما هي.

مثال

set_bit.cpp
int x = 4;      // 0100
x |= (1 << 0);  // 0100 | 0001 = 0101 (5)
x = 5

تصفير البِت رقم k (جعله 0)

x &= ~(1 << k);

  • 1 << k → قناع فيه البِت رقم k يساوي 1.

  • ~(1 << k) → قناع فيه البِت رقم k يساوي 0، وجميع البِتّات الأخرى تساوي 1.

  • إجراء & مع هذا القناع يُصفّر البِت رقم k فقط، ويُبقي كل شيء آخر كما هو.

مثال

clear_bit.cpp
int x = 13;         // 1101
x &= ~(1 << 2);     // ~(0100) = ...11111011
                    // 1101 & 1011 = 1001 (9)
x = 9

قلب (تبديل) البِت رقم k

x ^= (1 << k);

العامل ^ (XOR) يقلب البِت عندما يكون القناع فيه 1:

Original bitXOR 1Result
011
110

مثال

toggle_bit.cpp
int x = 5;     // 0101
x ^= (1 << 2); // 0101 ^ 0100 = 0001 (1)
x = 1

نشاط: مراجعة سريعة للحيل الأربع

حدّد إن كانت كل عبارة صحيحة أم خاطئة.

التعبير (x >> k) & 1 يعطي 1 إذا كان البِت رقم k مساويًا لـ1.

x |= (1 << k) قد يُصفّر بعض البِتّات الأخرى في x.

لتصفير البِت رقم k نستخدم x &= ~(1 << k).

تطبيق x ^= (1 << k) مرتين متتاليتين على نفس k يعيد x إلى قيمته الأصلية.

إزالة أقل بِت معيّن (أقل 1 من اليمين)

هذه حيلة مشهورة جدًا:

x = x & (x - 1);

هذا السطر يُزيل (يُصفّر) أقل بِت معيّن — أي أول 1 من جهة اليمين — في x.

لماذا تعمل هذه الحيلة؟

فكّر ماذا يحدث للبِتّات عندما نطرح 1:

  • عند تنفيذ x - 1، نمشي من اليمين حتى نجد أول بِت قيمته 1.

  • هذا البِت الأقل (أقرب 1 لليمين) يتحوّل إلى 0.

  • كل البِتّات التي على يمينه (وكانت 0) تتحوّل إلى 1 (بسبب الاقتراض في الطرح).

إذًا حول أقل بِت معيّن، النمط يكون بهذا الشكل:

pattern
x:       .... 1 0 0 0 0
x - 1:   .... 0 1 1 1 1

عند إجراء AND:

and_pattern
x        = .... 1 0 0 0 0
x - 1    = .... 0 1 1 1 1
-------------------------
x&(x-1)  = .... 0 0 0 0 0

  • أقل بِت معيّن يصبح 0 لأنه 1 & 0.

  • كل البِتّات التي على يمينه كانت 0 في x، فتُعطي 0 & شيء = 0 بغض النظر عن x-1.

  • كل البِتّات التي على يساره لا تتأثر، لأن الطرح لم يغيّرها؛ فتبقى كما هي بعد AND.

مثال

remove_lowest.cpp
x           = 12 = 1100
x - 1       = 11 = 1011
x & (x - 1) =  8 = 1000
أُزيل أقل بِت معيّن (البِت رقم 2)، والنتيجة = 8

مختبر تفاعلي: شاهد الاقتراض والإزالة خطوة بخطوة

x & (x - 1) مع x = 12

شغّل المحاكاة لمشاهدة كيف يتحوّل x إلى x-1 (لاحظ البِتّات الملوّنة بالاقتراض)، ثم كيف تُنتج عملية AND الإزالة.

x = 12
x - 1 = 11
x & (x - 1)
اضغط "شغّل المحاكاة" للبدء.

فحص ما إذا كان x قوة للعدد 2

بافتراض أن x عدد موجب، نستخدم الحيلة السابقة مباشرة:

bool isPowerOfTwo = (x & (x - 1)) == 0;

لماذا تعمل هذه الحيلة؟

  • العدد الذي هو قوة للعدد 2 في الثنائي (1, 10, 100, 1000, …) يحتوي بِتًا واحدًا فقط معيّنًا.

  • عند تنفيذ x & (x - 1) نُزيل أقل بِت معيّن.

  • إذا كان هناك بِت واحد فقط معيّن، فبعد إزالته تصبح القيمة 0.

إذًا: x & (x - 1) == 0 ⇔ x لديه بِت واحد معيّن فقط (أي قوة للعدد 2).

أمثلة

xbinaryx & (x-1)النتيجة
810000000قوة للعدد 2 ✓
1211001000ليست 0 → ليست قوة للعدد 2

الحصول على أقل بِت معيّن فقط

int lowest = x & -x;

هنا -x هو المتمّم الثنائي (two's complement) للعدد x — تمامًا مثل ما تعلمناه في الفصل السابق (قلب كل البِتّات ثم إضافة 1).

الفكرة: التعبير x & -x يعزل أقل بِت معيّن (أول 1 من جهة اليمين) ويُصفّر جميع البِتّات الأخرى.

مثال

isolate_lowest.cpp
x                      = 12 = ...00001100
-x (two's complement)      = ...11110100
x & -x                     = ...00000100 = 4
أقل بِت معيّن في x يمثّل القيمة 4 (أي البِت رقم 2)

مختبر تفاعلي: عزل أقل بِت معيّن عبر المتمّم الثنائي

x & -x مع x = 12 (على 8 بِتّات)

شاهد كيف يُبنى -x بقلب البِتّات ثم إضافة 1 (تمامًا كما في الفصل السابق)، ثم كيف تعزل عملية AND أقل بِت معيّن.

x = 12
flip(x)
-x = flip(x) + 1
x & -x
اضغط "شغّل المحاكاة" للبدء.

نشاط: مراجعة سريعة للحيل المتقدمة

حدّد إن كانت كل عبارة صحيحة أم خاطئة.

x & (x - 1) يزيل أقل بِت معيّن في x.

إذا كانت x قوة للعدد 2، فإن x & (x - 1) لن تساوي 0 بالضرورة.

x & -x يعزل أقل بِت معيّن فقط ويُصفّر كل البِتّات الأخرى.

لعزل أقل بِت معيّن نحتاج -x، وهو ما يساوي flip(x) + 1.

اكتشف الخطأ: && ليست & !

انظر إلى هذه الدالة التي يُفترض أنها تفحص إن كان البِت رقم k في n مساويًا لـ1:

buggy_kth_bit.cpp
bool kthBitIsSet(int n, int k) {
    return (n >> k) && 1;   // !! هذا خطأ !!
}

هذا الكود لا يعمل بشكل صحيح. جرّبه مع n = 10 (أي 1010):

kالبِت الحقيقي رقم k في 1010(n >> k) && 1
00true (خطأ!)
11true
20true (خطأ!)
31true

لماذا؟ لأن && هو عامل منطقي (logical AND) وليس عامل بِتّي (bitwise AND). التعبير (n >> k) && 1 يعني: "هل n >> k غير صفر؟" — وهذا يكون صحيحًا طالما بقي أي بِت معيّن في أي مكان من n >> k، وليس بالضرورة البِت الأخير فقط.

الإصلاح: استخدم & (عامل بِتّي واحد) بدلًا من &&: return (n >> k) & 1;

هذا الخلط بين & و&& (وكذلك بين | و||) من أكثر الأخطاء شيوعًا عند البدء بحيل البِتّات — تأكّد دائمًا من استخدام العامل البِتّي عندما تتعامل مع بِتّات فردية.

هذه أهم وأشهر حيل التعامل مع البِتّات (bit manipulation) في البرمجة التنافسية — احفظها جيدًا فستُستخدم كثيرًا في مسائل الأقنعة الثنائية (bitmasks) القادمة.

تمارين منزلية

1) تتبّع متسلسل

لدينا x = 6 (أي 0110). طبّق التسلسل التالي بالترتيب وحدّد قيمة x بعد كل خطوة: (أ) ضبط البِت رقم 0، (ب) تصفير البِت رقم 1، (ج) قلب البِت رقم 2.

trace.cpp
int x = 6;          // 0110
x |= (1 << 0);      // 0110 | 0001 = 0111 = 7
x &= ~(1 << 1);     // 0111 & 1101 = 0101 = 5
x ^= (1 << 2);      // 0101 ^ 0100 = 0001 = 1
بعد (أ): x = 7 (0111) بعد (ب): x = 5 (0101) بعد (ج): x = 1 (0001)

2) اكتشف الخطأ في حيلة الإزالة

زميل لك كتب الكود التالي لإزالة أقل بِت معيّن من y = 12، لكنه أخطأ في اختيار العامل:

find_error2.cpp
int y = 12;
y = y | (y - 1);   // المفروض إزالة أقل بِت معيّن

ما الخطأ؟ وما الناتج الفعلي مقارنة بالناتج الصحيح؟

الخطأ هو استخدام | (OR) بدلًا من & (AND). عملية OR لا تُزيل أي بِت، بل على العكس قد تُضيف بِتّات جديدة.

y | (y - 1) = 1100 | 1011 = 1111 = 15 (غير صحيح) y & (y - 1) = 1100 & 1011 = 1000 = 8 (صحيح)

الإصلاح: استخدم y = y & (y - 1);

3) فحص قوى العدد 2

باستخدام حيلة x > 0 && (x & (x - 1)) == 0، حدّد أيّ من الأعداد التالية هي قوة للعدد 2: 16, 18, 32, 33, 64.

16 → قوة للعدد 2 (10000) 18 → ليست قوة للعدد 2 (10010) 32 → قوة للعدد 2 (100000) 33 → ليست قوة للعدد 2 (100001) 64 → قوة للعدد 2 (1000000)

قاموس المصطلحات

Maskقناع

عدد ثنائي مصمّم خصيصًا لعزل أو تعديل بِتّات محددة عند دمجه مع عامل بِتّي.

LSB (Least Significant Bit)أقل بِت أهمية

البِت الأقصى يمينًا في التمثيل الثنائي، ويمثّل قيمة 2^0.

Set a bitتعيين بِت

جعل قيمة بِت معيّن تساوي 1 باستخدام | مع قناع مناسب.

Clear a bitتصفير بِت

جعل قيمة بِت معيّن تساوي 0 باستخدام & مع قناع معكوس (~).

Toggle a bitقلب بِت

عكس قيمة بِت معيّن (من 0 إلى 1 أو العكس) باستخدام ^.

Lowest set bitأقل بِت معيّن

أول بِت من جهة اليمين قيمته 1 في التمثيل الثنائي للعدد.

Power of twoقوة للعدد 2

عدد يحتوي بِتًا واحدًا فقط معيّنًا في تمثيله الثنائي، مثل 1, 2, 4, 8, 16.

Logical AND (&&) vs Bitwise AND (&)AND منطقي مقابل AND بِتّي

&& يفحص "صفر أم لا" للقيمة كاملة، بينما & يعمل بِتًا بِتًا — خلطهما خطأ شائع وخطير.
