if (c == 'N') y++;

X: 0

⚠️ Error: Out of bounds!

👨‍💻 C++ Implementation

#include <iostream>
#include <string>
using namespace std;

int main() {
    string moves;
    cin >> moves;
    
    int x = 0, y = 0; // Starting position at origin (0,0)
    
    // Simulation Loop
    for (char c : moves) {
        if (c == 'N') y++;       // Move North
        else if (c == 'S') y--;  // Move South
        else if (c == 'E') x++;  // Move East
        else if (c == 'W') x--;  // Move West
    }
    
    // Print final coordinates
    cout << x << " " << y << endl;
    
    return 0;
}
