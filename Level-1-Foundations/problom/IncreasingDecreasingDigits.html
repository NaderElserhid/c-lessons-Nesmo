Increasing Decreasing Digits

هذا الشرح مقدم من منصة تويت كود

تعريف الرقم المطلوب

الرقم يكون صحيحاً إذا كان يتكون من 3 خانات أو أكثر، ولا يحتوي على الرقم 0، وتكون أرقامه متزايدة بشكل صارم حتى نقطة معينة، ثم تتناقص بشكل صارم حتى النهاية.

مختبر الفحص المتسلسل

أدخل قيمة البداية:

solution.cpp
#include <iostream>
#include <string>
using namespace std;

bool isBeautiful(long long n) {
    string s = to_string(n);
    if (s.length() < 3) return false;
    for (char c : s) if (c == '0') return false;

    int i = 0;
    // التزايد الصارم (على الأقل طوله 2)
    while (i + 1 < s.length() && s[i] < s[i+1]) i++;
    if (i == 0 || i == s.length() - 1) return false;

    // التناقص الصارم (حتى نهاية الرقم)
    while (i + 1 < s.length() && s[i] > s[i+1]) i++;
    
    return (i == s.length() - 1);
}

int main() {
    long long n;
    cin >> n;
    n++;
    while (!isBeautiful(n)) n++;
    cout << n;
    return 0;
}
