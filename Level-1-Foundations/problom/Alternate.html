يتم حساب مجموع الحدود حيث يتبع النمط التكرار: (جمع، جمع، طرح) للحدود المتسلسلة:

1 · n + 2 · (n - 1) - 3 · (n - 2) + 4 · (n - 3) + 5 · (n - 4) - 6 · (n - 5) + ... ± n · 1

مختبر التتبع المتسلسل

أدخل قيمة n:

solution.cpp
#include <iostream>
using namespace std;

int main() {
    long long n;
    cin >> n;

    long long total_sum = 0;
    for (int i = 1; i <= n; i++) {
        long long term = (long long)i * (n - i + 1);
        
        // النمط المتبع: جمع، جمع، طرح
        if (i % 3 == 0) {
            total_sum -= term;
        } else {
            total_sum += term;
        }
    }

    cout << total_sum << endl;
    return 0;
}
