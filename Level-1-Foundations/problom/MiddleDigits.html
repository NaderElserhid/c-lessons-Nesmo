وصف المسألة

يتم إعطاؤك رقمين طبيعيين (بعدد خانات فردي). المطلوب هو كتابة دالة تقوم بتبديل "الخانة الوسطى" بين الرقمين وطباعة القيم الجديدة.

مختبر تبديل الخانات

swap_middle.cpp
#include <iostream>
#include <string>
using namespace std;

// دالة لتبديل الخانات الوسطى باستخدام المرجع (Reference)
void SwapMidDigits(string &x, string &y) {
    int midX = x.length() / 2;
    int midY = y.length() / 2;

    // عملية التبديل
    char temp = x[midX];
    x[midX] = y[midY];
    y[midY] = temp;
}

int main() {
    string x, y;
    cin >> x >> y;

    // استدعاء الدالة
    SwapMidDigits(x, y);

    // طباعة القيم الجديدة
    cout << x << " " << y << endl;

    return 0;
}
