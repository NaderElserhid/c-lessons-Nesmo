#include <iostream>
using namespace std;

int main () {
    int k2, k3, k5, k6;
    cin >> k2 >> k3 >> k5 >> k6;

    // نأخذ أقل كمية متوفرة من (2، 5، 6) لتكوين 256
    int k_256 = min(min(k2, k5), k6);

    // المتبقي من 2 مع المتوفر من 3 لتكوين 32
    int k_32 = min(k3, k2 - k_256);

    long long solution = (long long)k_256 * 256 + (long long)k_32 * 32;
    cout << solution << endl;

    return 0;
}
