$CATEGORY: المستوى الأول C++/عبارة if

::سؤال 1: مفهوم if::ما وظيفة عبارة if في C++؟ {
=تنفيذ جزء من الكود فقط إذا تحقق شرط معيّن
~تعريف متغير جديد في البرنامج
~تكرار تنفيذ نفس الكود إلى الأبد
}

::سؤال 2: البنية الأساسية::في البنية if (condition) statement;، ما الذي يوضع بين القوسين؟ {
=الشرط (condition) الذي سيتم فحصه
~اسم المتغير الذي سيُطبع
~عدد مرات التكرار
}

::سؤال 3: نتيجة الشرط::إلى ماذا يُقيَّم الشرط (condition) داخل if؟ {
=إمّا true أو false
~إلى رقم صحيح فقط
~إلى نص (string) دائماً
}

::سؤال 4: عندما يكون الشرط صحيحاً::إذا كان شرط if يساوي true، ماذا يحدث؟ {
=يتم تنفيذ الأمر (statement) المرتبط بـ if
~يتوقف البرنامج فوراً
~يُعاد فحص الشرط مرة أخرى تلقائياً
}

::سؤال 5: عندما يكون الشرط خاطئاً بدون else::إذا كان شرط if يساوي false ولا يوجد else، ماذا يحدث؟ {
=يتم تجاهل الأمر، ويستمر البرنامج في التنفيذ بعد عبارة if
~يتوقف البرنامج عن العمل
~يُنفَّذ الأمر مرة واحدة رغم ذلك
}

::سؤال 6: عامل المقارنة \=\=::في الكود if (x \=\= 100)، ماذا يعني \=\= هنا؟ {
=مقارنة (هل x تساوي 100؟) وليست عملية إسناد
~إسناد قيمة 100 إلى x
~طباعة قيمة x على الشاشة
}

::سؤال 7: متى تُطبع الرسالة::في الكود if (x \=\= 100) cout << "x is 100";، متى تُطبع الرسالة؟ {
=فقط عندما تكون قيمة x تساوي 100 بالضبط
~في كل مرة يُشغَّل فيها البرنامج بغض النظر عن x
~فقط عندما تكون x أكبر من 100
}

::سؤال 8: تنفيذ أكثر من أمر::لتنفيذ أكثر من أمر واحد عند تحقق شرط if، ماذا نستخدم؟ {
=نجمعها بين الأقواس المعقوفة \{ \}
~نفصلها بفواصل عادية ,
~نكررها داخل أقواس عادية ( )
}

::سؤال 9: معاملة الكتلة المركّبة::كيف تُعامَل الكتلة المركّبة \{ \} داخل if؟ {
=كأنها أمر واحد
~كأنها عدة برامج منفصلة
~لا تُعامَل كأمر إطلاقاً
}

::سؤال 10: تأثير المسافات البادئة::هل تؤثّر المسافات البادئة (indentation) وتنسيق الأسطر على طريقة تنفيذ البرنامج؟ {
=لا، لا تؤثّر على التنفيذ إطلاقاً
~نعم، تغيّر نتيجة البرنامج دائماً
~نعم، لكن فقط داخل الحلقات
}

::سؤال 11: عدد الأوامر داخل الكتلة::في الكود if (x \=\= 100) \{ cout << "x is "; cout << x; \}، كم أمراً يوجد داخل الكتلة المركّبة؟ {
=أمران (cout الأول وcout الثاني)
~أمر واحد فقط
~ثلاثة أوامر
}

::سؤال 12: وظيفة else::ما وظيفة else في C++؟ {
=تحديد ما يحدث عندما يكون شرط if غير صحيح
~تكرار نفس الشرط مرة أخرى
~إلغاء عبارة if بالكامل من الكود
}

::سؤال 13: شرط تنفيذ else::متى يُنفَّذ الجزء الموجود بعد else؟ {
=فقط عندما يكون شرط if السابق له false
~دائماً، بغض النظر عن نتيجة الشرط
~فقط عندما يكون شرط if صحيحاً
}

::سؤال 14: تتبّع كود else::في الكود: if (x \=\= 100) cout << "x is 100"; else cout << "x is not 100";، لو كانت x تساوي 50، ماذا يُطبع؟ {
=x is not 100
~x is 100
~لا يُطبع أي شيء
}

::سؤال 15: وظيفة else if::ما الغرض من استخدام else if؟ {
=ربط عدّة شروط متتالية للاختيار من بين أكثر من بديلين
~تعريف دالة جديدة في البرنامج
~إنهاء البرنامج عند تحقق شرط معيّن
}

::سؤال 16: طبيعة else if::هل else if كلمة مفتاحية جديدة ومستقلة في C++؟ {
=لا، هي ببساطة else يتبعها if جديدة
~نعم، وهي مختلفة تماماً عن if وelse
~نعم، ولا يمكن استخدامها إلا مرة واحدة فقط
}

::سؤال 17: عدد الفروع المنفَّذة::في سلسلة else if، كم فرعاً يُنفَّذ عادةً عند تحقق أكثر من شرط في السلسلة؟ {
=فرع واحد فقط - أول شرط يتحقق
~كل الفروع التي تتحقق شروطها
~آخر فرع في السلسلة دائماً
}

::سؤال 18: تتبّع سلسلة else if (موجب)::في السلسلة: if (x > 0) ... else if (x < 0) ... else ...، لو كانت x تساوي 5، ماذا يُطبع؟ {
=x is positive
~x is negative
~x is 0
}

::سؤال 19: تتبّع سلسلة else if (سالب)::نفس السلسلة السابقة، لو كانت x تساوي -3، ماذا يُطبع؟ {
=x is negative
~x is positive
~x is 0
}

::سؤال 20: تتبّع سلسلة else if (صفر)::نفس السلسلة السابقة، لو كانت x تساوي 0، ماذا يُطبع؟ {
=x is 0
~x is positive
~x is negative
}

::سؤال 21: مثال الطقس - معتدل::في مثال الطقس: if (temp >\= 30) حار؛ else if (temp >\= 15) معتدل؛ else بارد؛ لو كانت temp تساوي 20، ماذا يُطبع؟ {
=الجو معتدل، يوم لطيف
~الجو حار، خذ الماء معك
~الجو بارد، البس معطفاً
}

::سؤال 22: مثال الطقس - حار::نفس مثال الطقس، لو كانت temp تساوي 35، ماذا يُطبع؟ {
=الجو حار، خذ الماء معك
~الجو معتدل، يوم لطيف
~الجو بارد، البس معطفاً
}

::سؤال 23: مثال الطقس - بارد::نفس مثال الطقس، لو كانت temp تساوي 5، ماذا يُطبع؟ {
=الجو بارد، البس معطفاً
~الجو حار، خذ الماء معك
~الجو معتدل، يوم لطيف
}

::سؤال 24: مثال الدرجات - B::في مثال الدرجات: score >\= 90 يعطي A؛ score >\= 75 يعطي B؛ score >\= 60 يعطي C؛ وإلا يعطي F؛ لو كانت score تساوي 80، ما التقدير؟ {
=جيد جداً (B)
~ممتاز (A)
~مقبول (C)
}

::سؤال 25: مثال الدرجات - راسب::نفس مثال الدرجات، لو كانت score تساوي 50، ما التقدير؟ {

~مقبول (C)

}

::سؤال 26: مثال المواصلات - المشي::في مثال المواصلات: distance < 1 يعني امشِ؛ distance <\= 5 يعني دراجة؛ وإلا حافلة أو سيارة؛ لو كانت distance تساوي 0.5، ماذا يُقترح؟ {
=المسافة قريبة، امشِ على قدميك
~استخدم الدراجة
~استخدم الحافلة أو السيارة
}

::سؤال 27: مثال المواصلات - الحافلة::نفس مثال المواصلات، لو كانت distance تساوي 10، ماذا يُقترح؟ {

::سؤال 28: مثال الفئة العمرية - مراهق::في مثال الفئة العمرية: age < 13 يعني طفل؛ age < 20 يعني مراهق؛ وإلا بالغ؛ لو كانت age تساوي 15، ما التصنيف؟ {

~طفل
~بالغ
}

::سؤال 29: التعبير المنطقي::ما اسم التعبير الذي قيمته الوحيدة إمّا true أو false، مثل الشرط داخل if؟ {

~تعبير حسابي (Arithmetic expression)
~تعبير نصي (String expression)
}

::سؤال 30: عبارة الاختيار::ما الاسم العام لعبارات مثل if وswitch التي تختار مساراً من بين عدّة مسارات؟ {

~عبارة تكرار (Loop statement)
